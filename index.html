SQL para crear las tablas

Copiá todo esto y pegalo en Supabase → SQL Editor → New query, luego ejecutalo.

create table if not exists cotizaciones (
  id bigserial primary key,
  sb_id bigint unique not null,
  fecha date not null,
  insumo text not null,
  codigo text not null,
  proveedor text not null,
  cuenta text default '',
  cuit text default '',
  precio numeric(12,4) not null,
  moneda text default '$',
  score_entregas int default 0,
  score_precio int default 0,
  score_disponibilidad int default 0,
  score_plazo int default 0,
  total int default 0,
  created_at timestamptz default now()
);

create table if not exists historico_precios (
  id bigserial primary key,
  codigo text not null,
  insumo text not null,
  proveedor text not null,
  fecha date not null,
  precio numeric(12,4) not null,
  moneda text default '$',
  created_at timestamptz default now()
);

alter table cotizaciones enable row level security;
alter table historico_precios enable row level security;

create policy "allow all" on cotizaciones for all using (true) with check (true);
create policy "allow all" on historico_precios for all using (true) with check (true);
